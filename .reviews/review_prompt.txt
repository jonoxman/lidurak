# T-14: Frontend WebSocket Client & Interactive Game Play

## Objective
Implement the WebSocket connection manager, game state management, card interaction UX, and a working local play flow where the user can create a game via the REST API and play it interactively at `http://localhost:5173`.

## Requirements

### File: `frontend/src/api/client.ts`

REST API client:
```typescript
export async function createGame(playerCount: number, playerNames: string[], allowRedirect?: boolean): Promise<{ gameId: string, players: string[] }>
export async function getGameView(gameId: string, playerId: string): Promise<PlayerGameView>
export async function sendAction(gameId: string, action: GameAction): Promise<PlayerGameView | { error: string }>
```

### File: `frontend/src/api/websocket.ts`

WebSocket connection manager:
```typescript
export class GameWebSocket {
  constructor(gameId: string, playerId: string, onStateUpdate: (view: PlayerGameView) => void)
  connect(): void
  sendAction(action: GameAction): void
  disconnect(): void
}
```
- Connects to `ws://${location.host}/ws/game/${gameId}/${playerId}`
- Parses incoming `PlayerGameView` JSON and calls `onStateUpdate`
- Sends `GameAction` JSON objects
- Auto-reconnect on disconnect (with exponential backoff, max 3 retries)

### File: `frontend/src/types.ts`

TypeScript interfaces matching the backend JSON codecs:
```typescript
export interface Card { suit: string; rank: string }
export interface TablePair { attack: Card; defense?: Card }
export interface PublicPlayerSummary { id: string; name: string; cardCount: number; hasPassed: boolean; finishRank?: number }
export interface PlayerGameView {
  myId: string; myHand: Card[]; opponents: PublicPlayerSummary[];
  tablePairs: TablePair[]; talonCount: number; bottomTrumpCard?: Card;
  trumpSuit: string; attackerIdx: number; defenderIdx: number;
  boutPhase: string; legalActions: string[]; finishOrder: string[];
  isDraw: boolean; maxBoutAttacks: number;
}
export interface GameAction { type: string; playerId: string; [key: string]: any }
```

+export interface CreatedGame {
+  gameId: string;
+  players: string[];
+}
+
+async function readJson(response: Response): Promise<unknown> {
+  try {
+    return await response.json();
+  } catch {
+    return null;
+  }
+}
+
+function errorMessage(body: unknown, fallback: string): string {
+  if (typeof body === 'object' && body !== null) {
+    const error = (body as Record<string, unknown>).error;
+    if (typeof error === 'string' && error.length > 0) return error;
+  }
+  return fallback;
+}
+
+export async function createGame(
+  playerCount: number,
+  playerNames: string[],
+  allowRedirect = true,
+): Promise<CreatedGame> {
+  const response = await fetch('/api/game/create', {
+    method: 'POST',
+    headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ playerCount, playerNames, allowRedirect }),
+  });
+  const body = await readJson(response);
+
+  if (!response.ok) {
+    throw new Error(errorMessage(body, `Could not create game (HTTP ${response.status})`));
+  }
+  if (typeof body !== 'object' || body === null) {
+    throw new Error('Game creation response was not a JSON object');
+  }
+
+  const record = body as Record<string, unknown>;
+  const gameId = record.gameId;
+  if (typeof gameId !== 'string') {
+    throw new Error('Game creation response did not include a gameId');
+  }
+
+  const players = Array.isArray(record.players)
+    ? record.players
+        .map((player) => {
+          if (typeof player === 'string') return player;
+          if (typeof player === 'object' && player !== null) {
+            const id = (player as Record<string, unknown>).id;
+            if (typeof id === 'string') return id;
+          }
+          return '';
+        })
+        .filter((id) => id.length > 0)
+    : [];
+
+  return { gameId, players };
+}
+
+export async function getGameView(gameId: string, playerId: string): Promise<PlayerGameView> {
+  const response = await fetch(
+    `/api/game/${encodeURIComponent(gameId)}/view/${encodeURIComponent(playerId)}`,
+  );
+  const body = await readJson(response);
+
+  if (!response.ok) {
+    throw new Error(errorMessage(body, `Could not load the game view (HTTP ${response.status})`));
+  }
+  return parsePlayerGameView(body);
+}

+export async function sendAction(
+  gameId: string,
+  action: GameAction,
+): Promise<PlayerGameView | { error: string }> {
+  const response = await fetch(`/api/game/${encodeURIComponent(gameId)}/action`, {
+    method: 'POST',

+  const body = await readJson(response);
+
+  if (!response.ok) {
+    return { error: errorMessage(body, `Action failed (HTTP ${response.status})`) };
+  }
+  try {
+    return parsePlayerGameView(body);
+  } catch (reason: unknown) {
+    return { error: reason instanceof Error ? reason.message : String(reason) };
+  }

+import { parsePlayerGameView, type GameAction, type PlayerGameView } from '../types';
+
+/**
+ * Live connection to a game room. Incoming JSON is validated into a
+ * `PlayerGameView` before it reaches the UI, and a dropped connection is
+ * retried up to three times with exponential backoff.
+ */
+export class GameWebSocket {
+  private socket: WebSocket | null = null;
+  private retries = 0;
+  private closedByUser = false;
+  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
+  private readonly maxRetries = 3;
+
+  constructor(
+    private readonly gameId: string,
+    private readonly playerId: string,
+    private readonly onStateUpdate: (view: PlayerGameView) => void,
+  ) {}
+
+  connect(): void {
+    this.closedByUser = false;
+    // An explicit connect starts a fresh backoff budget, so a caller can still
+    // reconnect after auto-reconnect has exhausted its retries.
+    this.retries = 0;
+    if (this.reconnectTimer != null) {
+      clearTimeout(this.reconnectTimer);
+      this.reconnectTimer = null;
+    }
+    const socket = this.socket;
+    if (
+      socket != null &&
+      (socket.readyState === WebSocket.OPEN || socket.readyState === WebSocket.CONNECTING)
+    ) {
+      return;
+    }
+    this.socket = null;
+    this.open();
+  }
+
+  sendAction(action: GameAction): void {
+    if (this.isOpen()) {
+      this.socket!.send(JSON.stringify(action));
+    }
+  }
+
+  isOpen(): boolean {
+    return this.socket != null && this.socket.readyState === WebSocket.OPEN;
+  }
+
+  disconnect(): void {
+    this.closedByUser = true;
+    if (this.reconnectTimer != null) {
+      clearTimeout(this.reconnectTimer);
+      this.reconnectTimer = null;
+    }
+    const socket = this.socket;
+    this.socket = null;
+    if (socket != null) {
+      socket.close();
+    }
+  }
+
+  private open(): void {
+    const host = typeof location === 'undefined' ? 'localhost:8080' : location.host;
+    const scheme =
+      typeof location !== 'undefined' && location.protocol === 'https:' ? 'wss:' : 'ws:';
+    const gameId = encodeURIComponent(this.gameId);
+    const playerId = encodeURIComponent(this.playerId);
+    const socket = new WebSocket(`${scheme}//${host}/ws/game/${gameId}/${playerId}`);
+    this.socket = socket;
+
+    socket.onmessage = (event: MessageEvent) => {
+      try {
+        const view = parsePlayerGameView(JSON.parse(String(event.data)));
+        // A working connection resets the backoff budget.
+        this.retries = 0;
+        this.onStateUpdate(view);
+      } catch {
+        // Ignore frames that are not valid game views.
+      }
+    };
+
+    socket.onclose = () => {
+      // Ignore closes from sockets that a newer connect()/disconnect() replaced,
+      // otherwise the stale socket schedules a duplicate reconnect.
+      if (this.socket !== socket) return;
+      this.socket = null;
+      this.scheduleReconnect();
+    };
+
+    socket.onerror = () => {
+      // The browser closes the socket after an error; `onclose` handles retries.
+    };
+  }
+
+  private scheduleReconnect(): void {
+    if (this.closedByUser || this.retries >= this.maxRetries) return;
+    const delay = 1000 * 2 ** this.retries;
+    this.retries += 1;
+    this.reconnectTimer = setTimeout(() => {
+      this.reconnectTimer = null;
+      if (!this.closedByUser) this.open();
+    }, delay);
+  }
+}
