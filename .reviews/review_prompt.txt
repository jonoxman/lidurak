# T-13: SVG Playing Card Components & Responsive Table Layout

## Objective
Build custom inline SVG playing card components and the responsive game table layout using Preact, CSS Grid/Flexbox, and the Lichess dark-mode aesthetic.

## Requirements

### File: `frontend/src/components/Card.tsx`

A Preact component that renders an inline SVG playing card:
- **Props:** `{ suit: string, rank: string, faceUp?: boolean, selected?: boolean, onClick?: () => void }`
- **Face-up card:** White/cream rounded rectangle with rank text (top-left, bottom-right) and suit symbol (center). Suit colors: `♠♣` = `#bababa`, `♥♦` = `#c33`.
- **Face-down card:** Dark patterned rectangle (use `#262421` with a subtle pattern or cross-hatch).
- **Selected state:** Slight upward translation (`translateY(-8px)`) and green border highlight (`#629924`).
- Card dimensions: approximately `70×100px`, scalable via CSS.

### File: `frontend/src/components/CardBack.tsx`

A simpler component for opponent card backs (no rank/suit, just the back design).

### File: `frontend/src/components/TablePair.tsx`

Renders a single attack–defense pair:
- Attack card flat, defense card overlapping at a slight offset (translate ~20px right and ~15px down).
- If no defense card, show only the attack card.

### File: `frontend/src/components/GameTable.tsx`

The main game table layout:
- **Center:** Battle area showing `TablePair` components in a flex row/wrap.
- **Bottom:** Player's own hand — fanned/sorted cards, clickable for selection.
- **Top / Sides:** Opponent areas showing card back count and player name/status.
- **Top-Right:** Talon display — stacked card backs with count badge, plus the perpendicular bottom trump card (a card rotated 90°) visible while talon > 0.
- **Trump indicator:** Badge showing trump suit symbol and color.
- **Action buttons area:** Below the hand — contextual buttons (rendered by parent, this component provides the layout slot).

### File: `frontend/src/components/ActionBar.tsx`

Action buttons component:
- **Props:** `{ legalActions: string[], onAction: (action: object) => void }`
- Buttons: "Pass" (green), "Surrender" (red), "Redirect" (yellow/amber).
- Only show buttons relevant to `legalActions`.
- Styled with Lichess aesthetic: rounded, high-contrast, hover effects.

### File: `frontend/src/styles/table.css`

- Dark theme: `#161512` background, `#262421` panels, `#bababa` text.
- Responsive layout: CSS Grid for table regions, flexbox for card rows.
- Media queries for mobile (`< 640px`): stack opponent areas above battle area, reduce card size.
- Active turn indicator: `#629924` glow/border on the current attacker/defender.

+  onAction: (action: { type: string }) => void;

+interface ActionButton {
+  action: string;
+  type: string;
+  label: string;
+  variant: 'green' | 'red' | 'amber';
+}
+
+const BUTTONS: ActionButton[] = [
+  { action: 'pass', type: 'passTurn', label: 'Pass', variant: 'green' },
+  { action: 'surrender', type: 'declareSurrender', label: 'Surrender', variant: 'red' },
+  { action: 'redirect', type: 'redirect', label: 'Redirect', variant: 'amber' },
+];
+
+/** Contextual bout buttons; only actions present in `legalActions` are shown. */
+export function ActionBar({ legalActions, onAction }: ActionBarProps) {
+  const buttons = BUTTONS.filter((button) => legalActions.includes(button.action));
+
+  if (buttons.length === 0) return null;
+
+  return (
+    <div class="action-bar">
+      {buttons.map((button) => (
+        <button
+          key={button.action}
+          type="button"
+          class={`action-button ${button.variant}`}
+          onClick={() => onAction({ type: button.type })}
+        >
+          {button.label}
+        </button>
+      ))}
+    </div>
+  );

+export interface CardData {

+export interface CardProps {
+  suit: string;
+  rank: string;
+  faceUp?: boolean;
+  selected?: boolean;
+  onClick?: () => void;
+}
+
+const SUIT_SYMBOLS: Record<string, string> = {
+  spades: '♠',
+  hearts: '♥',
+  diamonds: '♦',
+  clubs: '♣',
+};
+
+const RANK_LABELS: Record<string, string> = {
+  two: '2',
+  three: '3',
+  four: '4',
+  five: '5',
+  six: '6',
+  seven: '7',
+  eight: '8',
+  nine: '9',
+  ten: '10',
+  jack: 'J',
+  queen: 'Q',
+  king: 'K',
+  ace: 'A',
+};
+
+export function isRedSuit(suit: string): boolean {
+  const normalized = suit.toLowerCase();
+  return normalized === 'hearts' || normalized === 'diamonds';
+}
+
+export function suitColor(suit: string): string {
+  return isRedSuit(suit) ? '#c33' : '#bababa';
+}
+
+export function suitSymbol(suit: string): string {
+  return SUIT_SYMBOLS[suit.toLowerCase()] ?? '?';
+}
+
+export function rankLabel(rank: string): string {
+  return RANK_LABELS[rank.toLowerCase()] ?? rank;
+}
+
+/** An inline SVG playing card, roughly 70x100px and scalable via CSS. */
+export function Card({ suit, rank, faceUp = true, selected = false, onClick }: CardProps) {
+  const classes = ['card-slot'];
+  if (selected) classes.push('selected');
+  if (onClick) classes.push('clickable');
+
+  if (!faceUp) {
+    return (
+      <div
+        class={classes.join(' ')}
+        data-selected={selected ? 'true' : 'false'}
+        onClick={onClick}
+      >
+        <CardBack />
+      </div>
+    );
+  }
+
+  const color = suitColor(suit);
+  const symbol = suitSymbol(suit);
+  const label = rankLabel(rank);
+
+  return (
+    <div
+      class={classes.join(' ')}
+      data-selected={selected ? 'true' : 'false'}
+      data-suit={suit.toLowerCase()}
+      data-rank={rank.toLowerCase()}

+export interface GameTableProps {
+  myId: string;
+  myHand: CardData[];
+  opponents: OpponentSummary[];
+  tablePairs: TablePairData[];
+  talonCount: number;
+  bottomTrumpCard?: CardData | null;

+  onCardClick?: (card: CardData) => void;
+  children?: ComponentChildren;
+}
+
+const SUIT_ORDER: Record<string, number> = {
+  spades: 0,
+  hearts: 1,
+  diamonds: 2,
+  clubs: 3,
+};
+
+const RANK_ORDER: Record<string, number> = {
+  two: 2,
+  three: 3,
+  four: 4,
+  five: 5,
+  six: 6,
+  seven: 7,
+  eight: 8,
+  nine: 9,
+  ten: 10,
+  jack: 11,
+  queen: 12,
+  king: 13,
+  ace: 14,
+};
+
+export const cardKey = (card: CardData): string =>
+  `${card.suit.toLowerCase()}:${card.rank.toLowerCase()}`;
+
+const sortHand = (hand: CardData[]): CardData[] =>
+  [...hand].sort((left, right) => {
+    const suitDiff =
+      (SUIT_ORDER[left.suit.toLowerCase()] ?? 9) -
+      (SUIT_ORDER[right.suit.toLowerCase()] ?? 9);
+    if (suitDiff !== 0) return suitDiff;
+    return (
+      (RANK_ORDER[left.rank.toLowerCase()] ?? 0) -
+      (RANK_ORDER[right.rank.toLowerCase()] ?? 0)
+    );
+  });
+
+/**
+ * The responsive game table: opponents and talon on top, the battle area in the
+ * middle, the viewer's sorted hand at the bottom, and an action slot below it.
+ */
+export function GameTable({
+  myId,
+  myHand,
+  opponents,
+  tablePairs,
+  talonCount,
+  bottomTrumpCard,
+  trumpSuit,
+  attackerId,
+  defenderId,
+  selectedKeys = [],
+  onCardClick,
+  children,
+}: GameTableProps) {
+  const hand = sortHand(myHand);
+
+  return (
+    <div class="game-table" data-my-id={myId}>
+      <header class="table-top">
+        <section class="opponents" aria-label="Opponents">
+          {opponents.map((opponent) => {
+            const classes = ['opponent'];
+            if (opponent.id === attackerId) classes.push('active-attacker');
+            if (opponent.id === defenderId) classes.push('active-defender');
+            return (
+              <div key={opponent.id} class={classes.join(' ')} data-testid="opponent">
+                <span class="opponent-name">{opponent.name}</span>
+                <div class="opponent-cards" aria-label={`${opponent.name}'s cards`}>
+                  {Array.from({ length: opponent.cardCount }, (_, index) => (
+                    <CardBack key={index} />
+                  ))}
+                </div>
+                <span class="opponent-count">{opponent.cardCount} cards</span>
+                {opponent.finishRank != null ? (
+                  <span class="opponent-rank">Finished #{opponent.finishRank}</span>
+                ) : null}
+                {opponent.hasPassed ? <span class="opponent-passed">Passed</span> : null}
+              </div>
+            );
+          })}
+        </section>
+
+        <aside class="table-status" aria-label="Talon">

+            <div class="bottom-trump" aria-label="Bottom trump card">
+              <div class="bottom-trump-card">
+                <Card suit={bottomTrumpCard.suit} rank={bottomTrumpCard.rank} />
+              </div>
+            </div>
+          ) : null}
+          <div
+            class={`trump-badge ${isRedSuit(trumpSuit) ? 'red' : 'black'}`}
+            data-suit={trumpSuit.toLowerCase()}
+          >
+            <span>Trump</span>

+          </div>
+        </aside>
+      </header>
+
+      <section class="battle" aria-label="Battle">
+        {tablePairs.length === 0 ? (

+          ))
+        )}
+      </section>
+
+      <section class="hand" aria-label="Your hand">

+            suit={card.suit}

+        ))}

+
+      <footer class="actions-slot" aria-label="Actions">
+        {children}
+      </footer>
